const express = require('express');
const fs = require('fs');
const path = require('path');
const multer = require('multer');

const app = express();
const PORT = process.env.PORT || 3000;

// Configurar multer para subir imágenes
const storage = multer.diskStorage({
  destination: (req, file, cb) => {
    const uploadDir = path.join(__dirname, 'public/uploads');
    if (!fs.existsSync(uploadDir)) {
      fs.mkdirSync(uploadDir, { recursive: true });
    }
    cb(null, uploadDir);
  },
  filename: (req, file, cb) => {
    cb(null, 'profile-' + Date.now() + path.extname(file.originalname));
  }
});

const upload = multer({ 
  storage,
  fileFilter: (req, file, cb) => {
    if (file.mimetype.startsWith('image/')) {
      cb(null, true);
    } else {
      cb(new Error('Solo se permiten imágenes'), false);
    }
  }
});

app.use(express.json());
app.use(express.static('public'));

const dataPath = path.join(__dirname, 'data.json');

if (!fs.existsSync(dataPath)) {
  const initialData = {
    profile: { photo: "https://i.pravatar.cc/150?img=47", name: "TU NOMBRE", description: "Tu descripción" },
    design: {
      bgColor: "#0a0a0a",
      glowColor: "#cc0000",
      glowOpacity: 0.55,
      accentColor: "#cc0000",
      textColor: "#ffffff"
    },
    links: [
      { id: "1", label: "Mi Producto", url: "", emoji: "🛍️", order: 1 },
      { id: "2", label: "Quiz", url: "", emoji: "📝", order: 2 },
      { id: "3", label: "YouTube", url: "", emoji: "▶️", order: 3 },
      { id: "4", label: "Instagram", url: "", emoji: "📸", order: 4 },
      { id: "5", label: "WhatsApp", url: "", emoji: "💬", order: 5 }
    ],
    clicks: {}
  };
  fs.writeFileSync(dataPath, JSON.stringify(initialData, null, 2));
}

app.get('/api/data', (req, res) => {
  const data = JSON.parse(fs.readFileSync(dataPath, 'utf8'));
  res.json(data);
});

app.post('/api/data', (req, res) => {
  const data = JSON.parse(fs.readFileSync(dataPath, 'utf8'));
  data.profile = req.body.profile;
  data.design = req.body.design;
  data.links = req.body.links;
  fs.writeFileSync(dataPath, JSON.stringify(data, null, 2));
  res.json({ success: true });
});

app.post('/api/upload', upload.single('photo'), (req, res) => {
  if (!req.file) {
    return res.status(400).json({ error: 'No file uploaded' });
  }
  const photoPath = `/uploads/${req.file.filename}`;
  res.json({ photo: photoPath });
});

app.post('/api/track/:id', (req, res) => {
  const data = JSON.parse(fs.readFileSync(dataPath, 'utf8'));
  const linkId = req.params.id;
  if (!data.clicks[linkId]) data.clicks[linkId] = 0;
  data.clicks[linkId]++;
  fs.writeFileSync(dataPath, JSON.stringify(data, null, 2));
  res.json({ clicks: data.clicks[linkId] });
});

app.listen(PORT, () => console.log(`✅ Server on port ${PORT}`));

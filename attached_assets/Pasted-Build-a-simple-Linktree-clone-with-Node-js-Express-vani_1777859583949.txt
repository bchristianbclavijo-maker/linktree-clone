Build a simple Linktree clone with Node.js + Express + vanilla JavaScript.

## Files to create:

### 1. server.js
```javascript
const express = require('express');
const fs = require('fs');
const path = require('path');

const app = express();
const PORT = process.env.PORT || 3000;

app.use(express.json());
app.use(express.static('public'));

const dataPath = path.join(__dirname, 'data.json');

if (!fs.existsSync(dataPath)) {
  const initialData = {
    profile: { photo: "https://i.pravatar.cc/150?img=47", name: "TU NOMBRE", description: "Tu descripción" },
    links: [
      { id: "1", label: "Mi Producto", url: "", emoji: "🛍️", order: 1 },
      { id: "2", label: "Quiz", url: "", emoji: "📝", order: 2 },
      { id: "3", label: "YouTube", url: "", emoji: "▶️", order: 3 },
      { id: "4", label: "Instagram", url: "", emoji: "📸", order: 4 },
      { id: "5", label: "WhatsApp", url: "", emoji: "💬", order: 5 }
    ],
    clicks: {}
  };
  fs.writeFileSync(dataPath, JSON.stringify(initialData, null, 2));
}

app.get('/api/data', (req, res) => {
  const data = JSON.parse(fs.readFileSync(dataPath, 'utf8'));
  res.json(data);
});

app.post('/api/data', (req, res) => {
  const data = JSON.parse(fs.readFileSync(dataPath, 'utf8'));
  data.profile = req.body.profile;
  data.links = req.body.links;
  fs.writeFileSync(dataPath, JSON.stringify(data, null, 2));
  res.json({ success: true });
});

app.post('/api/track/:id', (req, res) => {
  const data = JSON.parse(fs.readFileSync(dataPath, 'utf8'));
  const linkId = req.params.id;
  if (!data.clicks[linkId]) data.clicks[linkId] = 0;
  data.clicks[linkId]++;
  fs.writeFileSync(dataPath, JSON.stringify(data, null, 2));
  res.json({ clicks: data.clicks[linkId] });
});

app.listen(PORT, () => console.log(`✅ Server on port ${PORT}`));
```

### 3. public/index.html
```html
<!DOCTYPE html>
<html lang="es">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Mis Links</title>
  <link href="https://fonts.googleapis.com/css2?family=Bebas+Neue&family=Montserrat:wght@300;400&display=swap" rel="stylesheet">
  <link rel="stylesheet" href="style.css">
</head>
<body>
  <div class="background-glow"></div>
  <div class="grid-overlay"></div>
  
  <div class="container">
    <div class="profile">
      <img id="profilePhoto" src="" alt="Foto" class="profile-photo">
      <h1 id="profileName"></h1>
      <p id="profileDesc" class="description"></p>
    </div>
    
    <div id="linksContainer" class="links-container"></div>
  </div>

  <script>
    async function loadPage() {
      const res = await fetch('/api/data');
      const data = await res.json();
      
      document.getElementById('profilePhoto').src = data.profile.photo;
      document.getElementById('profileName').textContent = data.profile.name;
      document.getElementById('profileDesc').textContent = data.profile.description;
      
      data.links
        .filter(link => link.url)
        .sort((a, b) => a.order - b.order)
        .forEach(link => {
          const btn = document.createElement('a');
          btn.href = link.url;
          btn.target = '_blank';
          btn.className = 'link-btn';
          btn.innerHTML = `<span>${link.emoji || '🔗'}</span> ${link.label}`;
          btn.onclick = () => fetch(`/api/track/${link.id}`, { method: 'POST' });
          document.getElementById('linksContainer').appendChild(btn);
        });
    }
    loadPage();
  </script>
</body>
</html>
```

  <script>
    let data = {};

    function checkAuth() {
      if (localStorage.getItem('adminAuth')) showAdmin();
    }

    function login() {
      if (document.getElementById('passwordInput').value === 'admin123') {
        localStorage.setItem('adminAuth', 'true');
        showAdmin();
      } else alert('❌ Contraseña incorrecta');
    }

    function logout() {
      localStorage.removeItem('adminAuth');
      location.reload();
    }

    function showAdmin() {
      document.getElementById('loginScreen').style.display = 'none';
      document.getElementById('adminPanel').style.display = 'block';
      loadData();
    }

    async function loadData() {
      const res = await fetch('/api/data');
      data = await res.json();
      document.getElementById('photoUrl').value = data.profile.photo;
      document.getElementById('nameInput').value = data.profile.name;
      document.getElementById('descInput').value = data.profile.description;
      renderLinks();
      renderAnalytics();
    }

    function renderLinks() {
      const container = document.getElementById('linksContainer');
      container.innerHTML = '';
      data.links.forEach((link, idx) => {
        const row = document.createElement('div');
        row.className = 'link-row';
        row.innerHTML = `
          <input type="text" placeholder="Label" value="${link.label}" onchange="data.links[${idx}].label = this.value" />
          <input type="url" placeholder="URL" value="${link.url}" onchange="data.links[${idx}].url = this.value" />
          <input type="text" placeholder="😊" value="${link.emoji || '🔗'}" maxlength="2" onchange="data.links[${idx}].emoji = this.value" />
          <button class="delete-btn" onclick="data.links.splice(${idx}, 1); renderLinks()">×</button>
        `;
        container.appendChild(row);
      });
    }

    function addLink() {
      data.links.push({ id: Date.now().toString(), label: 'Nuevo', url: '', emoji: '🔗', order: data.links.length + 1 });
      renderLinks();
    }

    function renderAnalytics() {
      const total = Object.values(data.clicks || {}).reduce((a, b) => a + b, 0);
      document.getElementById('analyticsContainer').innerHTML = `
        <div class="stat-box">
          <div class="stat-number">${total}</div>
          <div class="stat-label">Clics totales</div>
        </div>
      `;
    }

    async function saveChanges() {
      data.profile.photo = document.getElementById('photoUrl').value;
      data.profile.name = document.getElementById('nameInput').value;
      data.profile.description = document.getElementById('descInput').value;
      await fetch('/api/data', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) });
      document.getElementById('successMsg').style.display = 'block';
      setTimeout(() => document.getElementById('successMsg').style.display = 'none', 2000);
    }

    checkAuth();
  </script>
